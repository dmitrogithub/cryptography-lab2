cmake_minimum_required(VERSION 3.16)
project(kupyna_lab VERSION 1.0 LANGUAGES CXX)
add_executable(kupyna-lab src/main.cpp src/kupyna.cpp tests/selftest.cpp)
target_compile_features(kupyna-lab PRIVATE cxx_std_17)
target_include_directories(kupyna-lab PRIVATE src tests)
if(MSVC)
  target_compile_options(kupyna-lab PRIVATE /W4 /utf-8 /EHsc)
  set_property(TARGET kupyna-lab PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(kupyna-lab PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(kupyna-lab PRIVATE -municode)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_test(NAME kupyna-selftest COMMAND kupyna-lab selftest)
endif()
